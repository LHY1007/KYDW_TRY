def load_digit_data():  # 按可用数据源读取手写数字图像与标签
    """返回 train_dataset, test_dataset, dataset_name。图像范围统一为 [0, 1]，形状为 [1, 28, 28]。"""
    # 输入：可能存在的 CSV；输出：train_dataset、test_dataset、dataset_name。
    # Kaggle Digit Recognizer 常见路径
    csv_candidates = list(Path('/kaggle/input').glob('**/train.csv')) if Path('/kaggle/input').exists() else []
    for csv_path in csv_candidates:
        try:
            df = pd.read_csv(csv_path)
            pixel_cols = [c for c in df.columns if str(c).startswith('pixel')]  # 选出表示 784 个图像像素的列
            if 'label' in df.columns and len(pixel_cols) == 784:
                x = torch.tensor(df[pixel_cols].to_numpy(), dtype=torch.float32).reshape(-1, 1, 28, 28) / 255.0
                y = torch.tensor(df['label'].to_numpy(), dtype=torch.long)
                g = torch.Generator().manual_seed(SEED)  # 固定随机状态以便复现实验
                n_test = max(1000, int(0.15 * len(x)))  # 确定保留作测试的样本数量
                perm = torch.randperm(len(x), generator=g)  # 生成可复现的随机样本顺序
                test_idx, train_idx = perm[:n_test], perm[n_test:]  # 把打乱的样本索引分为测试和训练两组
                return TensorDataset(x[train_idx], y[train_idx]), TensorDataset(x[test_idx], y[test_idx]), 'MNIST / Kaggle Digit Recognizer'
        except Exception:
            continue  # 继续尝试其他数据来源

    # 标准 torchvision MNIST
    try:
        tfm = transforms.ToTensor()
        root = WORKDIR / 'mnist_data'  # 指定 MNIST 的下载与缓存位置
        train_ds = datasets.MNIST(root=root, train=True, transform=tfm, download=True)  # 建立互相隔离的数据划分
        test_ds = datasets.MNIST(root=root, train=False, transform=tfm, download=True)  # 建立互相隔离的数据划分

    except Exception as exc:
        warnings.warn(f'MNIST 暂时无法读取，使用 sklearn digits 完成本地结构检查：{exc}')

任务 2：训练集归一化

归一化把像素变成以训练集为基准的数值，能让优化过程更稳定。验证集和测试集共同使用训练子集的均值和标准差。

计算与核对： 遍历 stat_loader，累计每批 images 的像素总和、平方和和像素数量，再计算 train_mean 与 train_std。先用 E[x²] - E[x]² 计算方差，再开平方得到标准差。

输入与输出： 每批输入 images 的 shape 是 [batch, 1, 28, 28]；输出 train_mean、train_std 是标量，normalize_batch(images) 返回同样 shape 的归一化张量。

检查： 均值和标准差只能来自训练子集；底部断言和后面的三个 DataLoader 创建应顺利运行。

# 参考实现：从训练子集计算像素均值与标准差
stat_loader = DataLoader(train_dataset, batch_size=256, shuffle=False)  # 按批次读取数据，训练集按设置打乱顺序
pixel_sum = 0.0  # 累计训练图像像素总和，用于计算均值
pixel_sq_sum = 0.0  # 累计训练图像像素平方和，用于计算方差
pixel_count = 0  # 累计参与统计的训练像素数量

for images, _ in stat_loader:
    # 参考实现：累计 images 的总和、平方和与像素数量
    pixel_sum += images.sum().item()  # 累计训练图像像素总和，用于计算均值
    pixel_sq_sum += (images ** 2).sum().item()  # 累计训练图像像素平方和，用于计算方差
    pixel_count += images.numel()  # 累计参与统计的训练像素数量

train_mean = pixel_sum / pixel_count  # 参考实现：
train_std = max(pixel_sq_sum / pixel_count - train_mean ** 2, 1e-8) ** 0.5   # 先由像素二阶矩计算方差，再开平方得到标准差
print('train_mean =', train_mean, 'train_std =', train_std)
assert 0 < train_mean < 1  # 核对数据或计算结果是否符合预期条件
assert train_std > 0  # 核对数据或计算结果是否符合预期条件

def normalize_batch(images):  # 用训练子集的均值和标准差归一化图像
    return (images - train_mean) / (train_std + 1e-8)

class SmallCNN(nn.Module):
    def __init__(self):
        super().__init__()
        self.features = nn.Sequential(
            nn.Conv2d(1, 16, kernel_size=3, padding=1),  # 用卷积提取局部图像特征
            nn.ReLU(),  # 加入非线性激活
            nn.MaxPool2d(2),  # 对局部区域取最大值并缩小空间尺寸
            # 参考实现：加入 16→32 的 3×3 卷积、ReLU 和 2×2 最大池化
            nn.Conv2d(16, 32, kernel_size=3, padding=1),  # 用卷积提取局部图像特征
            nn.ReLU(),  # 加入非线性激活
            nn.MaxPool2d(2),  # 对局部区域取最大值并缩小空间尺寸
        )
        self.classifier = nn.Sequential(
            nn.Flatten(),  # 把空间特征展平为向量
            # 参考实现：加入 32×7×7 → 64 的全连接层、ReLU，以及 64 → 10 的输出层
            nn.Linear(32 * 7 * 7, 64),  # 用线性层转换特征维数
            nn.ReLU(),  # 加入非线性激活
            nn.Linear(64, 10),  # 用线性层转换特征维数
        )

    def forward(self, x):
        x = self.features(x)
        return self.classifier(x)

任务 4：实现训练步骤

一个训练批次依次完成：清空旧梯度、前向传播、计算损失、反向传播、更新参数。每轮训练结束后，验证集只用于评价，不更新参数。

计算与核对： 参考代码按上述顺序执行五个训练动作。

输入与输出： 输入是归一化后的 images [batch, 1, 28, 28] 和 labels；输出是标量 loss、批次预测和训练/验证指标。随后会保存训练曲线。

criterion = nn.CrossEntropyLoss()  # 定义分类损失函数
optimizer = torch.optim.Adam(model.parameters(), lr=1e-3)  # 为模型参数配置优化器和学习率

def evaluate(model, loader, noise_sigma=0.0):  # 在指定数据集计算损失、准确率与预测标签
    model.eval()  # 切换为评价模式
    ys, ps = [], []  # 分别收集真实标签与预测标签
    total_loss = 0.0  # 累计评价集各批次的样本损失
    with torch.no_grad():  # 评价和展示时关闭梯度记录
        for images, labels in loader:
            images, labels = images.to(device), labels.to(device)  # 把图像与标签移动到同一计算设备
            if noise_sigma > 0:
                images = torch.clamp(images + noise_sigma * torch.randn_like(images), 0, 1)
            logits = model(normalize_batch(images))
            total_loss += criterion(logits, labels).item() * len(labels)  # 计算预测值与目标之间的损失
            preds = logits.argmax(dim=1)  # 取分数最高的类别作为预测标签
            ys.extend(labels.cpu().numpy())
            ps.extend(preds.cpu().numpy())
    return total_loss / len(loader.dataset), accuracy_score(ys, ps), np.array(ys), np.array(ps)

def train_one_epoch(model, loader, optimizer):  # 遍历训练批次并返回本轮平均损失和准确率
    model.train()  # 切换为训练模式
    running_loss = 0.0  # 累计本轮训练的样本损失
    correct = 0  # 累计预测正确的样本数
    total = 0  # 累计本轮处理的样本数
    for images, labels in loader:
        images, labels = images.to(device), labels.to(device)  # 把图像与标签移动到同一计算设备
        images = normalize_batch(images)

        # 参考实现：完成五个训练动作
        # 1. optimizer 清空梯度
        # 2. model 前向传播得到 logits
        # 3. criterion 计算 loss
        # 4. loss 反向传播
        # 5. optimizer 更新参数
        optimizer.zero_grad(set_to_none=True)  # 清空上一个批次留下的梯度
        logits = model(images)
        loss = criterion(logits, labels)  # 计算预测值与目标之间的损失
        loss.backward()  # 根据损失计算各参数的梯度
        optimizer.step()  # 按当前梯度更新模型参数

        running_loss += loss.item() * len(labels)  # 累计本轮训练的样本损失
        correct += (logits.argmax(dim=1) == labels).sum().item()  # 取分数最高的类别作为预测标签
        total += len(labels)  # 累计本轮处理的样本数
    return running_loss / total, correct / total
