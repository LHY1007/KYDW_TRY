步骤 4/10

切分图块与计算连续评分

RGB 转 HED 后读取苏木精相关通道，计算连续核相关评分。两个阈值只从训练原图计算。

参考答案

PATCH, SIZE = 256, 128
images, scores, source_ids, coordinates = [], [], [], []
for source_id, image in enumerate(originals):
    height, width = image.shape[:2]
    for y in range(0, height - PATCH + 1, PATCH):
        for x in range(0, width - PATCH + 1, PATCH):
            tile = image[y:y+PATCH, x:x+PATCH]
            small = np.asarray(Image.fromarray(tile).resize((SIZE, SIZE), Image.Resampling.LANCZOS))
            hematoxylin = np.clip(rgb2hed(small)[..., 0], 0, None)
            images.append(small)
            scores.append(float(hematoxylin.mean()))
            source_ids.append(source_id)
            coordinates.append((x, y))

images = np.asarray(images, dtype=np.uint8)
scores = np.asarray(scores, dtype=np.float32)
source_ids = np.asarray(source_ids)
coordinates = np.asarray(coordinates)
assert images.shape == (576, 128, 128, 3)

def color_features(batch):
    z = batch.astype(np.float32) / 255.0
    return np.c_[z.mean((1, 2)), z.std((1, 2))]

class PatchDataset(Dataset):
    def __init__(self, indices, augment=False):
        self.indices = np.asarray(indices)
        self.augment = augment
    def __len__(self):
        return len(self.indices)
    def __getitem__(self, k):
        i = int(self.indices[k])
        x = images[i].astype(np.float32) / 255.0
        if self.augment and random.random() < 0.5:
            x = np.fliplr(x).copy()
        return torch.from_numpy(x.transpose(2, 0, 1)), torch.tensor(labels[i]), i

class PatchCNN(nn.Module):
    def __init__(self):
        super().__init__()
        def block(cin, cout):
            return nn.Sequential(
                nn.Conv2d(cin, cout, 3, padding=1), nn.BatchNorm2d(cout), nn.ReLU(),
                nn.MaxPool2d(2)
            )
        self.features = nn.Sequential(block(3, 24), block(24, 48), block(48, 96), nn.AdaptiveAvgPool2d(1))
        self.classifier = nn.Linear(96, 3)
    def forward(self, x):
        return self.classifier(self.features(x).flatten(1))

model = PatchCNN().to(DEVICE)
步骤 8/10

训练与验证选模

每轮计算训练损失和验证宏平均 F1，验证结果提升时保存 state_dict。

参考答案

train_loader = DataLoader(PatchDataset(tr, True), 32, shuffle=True, num_workers=2)
val_loader = DataLoader(PatchDataset(va), 64, shuffle=False, num_workers=2)
test_loader = DataLoader(PatchDataset(te), 64, shuffle=False, num_workers=2)
loss_fn = nn.CrossEntropyLoss()
optimizer = torch.optim.Adam(model.parameters(), lr=1e-3)

def evaluate(loader):
    model.eval(); ys, ps, ids, losses = [], [], [], []
    with torch.no_grad():
        for x, y, idx in loader:
            logits = model(x.to(DEVICE))
            losses.append(loss_fn(logits, y.to(DEVICE)).item())
            ys.extend(y.numpy()); ps.extend(logits.argmax(1).cpu().numpy()); ids.extend(idx.numpy())
    return float(np.mean(losses)), accuracy_score(ys, ps), f1_score(ys, ps, average="macro"), np.array(ys), np.array(ps), np.array(ids)

history, best_state, best_f1 = [], None, -1
for epoch in range(5):
    model.train(); train_losses = []
    for x, y, _ in train_loader:
        optimizer.zero_grad()
        logits = model(x.to(DEVICE))
        loss = loss_fn(logits, y.to(DEVICE))
        loss.backward(); optimizer.step()
        train_losses.append(loss.item())
    val_loss, val_acc, val_f1, *_ = evaluate(val_loader)
    history.append([np.mean(train_losses), val_loss, val_f1])
    if val_f1 > best_f1:
        best_f1 = val_f1
        best_state = {k: v.detach().cpu().clone() for k, v in model.state_dict().items()}
model.load_state_dict(best_state)
